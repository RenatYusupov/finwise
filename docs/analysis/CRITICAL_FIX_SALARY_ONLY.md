## ✅ Исправление

### 1. Фильтрация доходов по типу

```typescript
function getMonthlySalary(
  transactions: Transaction[],
  months: number = 6
): number[] {
  const end = new Date();
  const salaries: number[] = [];
  
  for (let i = 0; i < months; i++) {
    const monthStart = new Date(end.getFullYear(), end.getMonth() - i - 1, 1);
    const monthEnd = new Date(end.getFullYear(), end.getMonth() - i, 0);
    
    // Фильтруем только зарплату
    const monthSalary = transactions
      .filter(t => 
        t.type === 'income' &&
        t.date >= monthStart.toISOString() &&
        t.date <= monthEnd.toISOString() &&
        isSalaryTransaction(t)  // <-- НОВАЯ ФУНКЦИЯ
      )
      .reduce((sum, t) => sum + t.amount, 0);
    
    if (monthSalary > 0) {
      salaries.push(monthSalary);
    }
  }
  
  return salaries;
}

function isSalaryTransaction(transaction: Transaction): boolean {
  // Проверяем категорию
  if (transaction.categoryId === 'salary') {
    return true;
  }
  
  // Проверяем описание
  const salaryKeywords = [
    'Зарплата',
    'Заработная плата',
    'Оплата Заработная плата',
    'Перевод начисления Зарплата',
  ];
  
  const description = transaction.description.toLowerCase();
  return salaryKeywords.some(keyword => 
    description.includes(keyword.toLowerCase())
  );
}
```
